# Quick Start

## Probe Cluster Endpoints

```crystal
require "crybase"

client = CryBase::CouchBase::Client.new("couchbase://node1,node2")

client.connect.each do |endpoint|
  puts endpoint
end

client.close
```

## Write And Read One KV Document

```crystal
require "crybase"

kv = CryBase::CouchBase::KV::Client.from_string(
  "couchbase://Administrator:password@127.0.0.1/commerce",
)

kv.set("crybase:hello", %({"hello":"world"}))
puts String.new(kv.get("crybase:hello"))

kv.scope = "ecommerce_shop"
kv.collection = "users"
kv.set("user:1", %({"name":"Ada"}))
puts String.new(kv.collection("users").get("user:1"))

kv.delete("crybase:hello")
kv.delete("user:1")
kv.close
```

## Run One Query Statement

```crystal
require "crybase"

query = CryBase::CouchBase::Query::Client.from_string(
  "couchbase://Administrator:password@127.0.0.1/commerce",
)

query.scope = "ecommerce_shop"

result = query.query(
  "SELECT META(u).id AS doc_key, u.name FROM users AS u LIMIT 5",
)

puts result.rows.first["name"].as_s
query.close
```
